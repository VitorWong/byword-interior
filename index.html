    // 语言切换

    let isEN = false;

    function toggleLang() {

      isEN = !isEN;

      document.querySelectorAll('[data-en]').forEach(el => {

        el.textContent = isEN ? el.dataset.en : el.dataset.zh || el.textContent;

      });

      document.querySelector('.lang-switch').textContent = isEN ? '中' : 'EN';

    }

 

    // 主题切换

    function toggleTheme() {

      const theme = document.body.dataset.theme === 'light' ? 'dark' : 'light';

      document.body.dataset.theme = theme;

      document.querySelector('.theme-switch').textContent = theme === 'dark' ? '☀️' : '🌙';
